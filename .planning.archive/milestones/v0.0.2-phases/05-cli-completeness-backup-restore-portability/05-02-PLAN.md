**`lib/glorbo/cli/lifecycle/daemon.ex`** (~30 lines):

```elixir
defmodule Glorbo.CLI.Lifecycle.Daemon do
  @moduledoc """
  Spawns a detached child process that outlives the calling BEAM.

  Uses `/usr/bin/setsid` (available on all Linux targets) to put the child
  in its own session so SIGHUP from the caller's controlling terminal doesn't
  propagate. Returns the child's OS pid.

  Per RESEARCH.md §Open Question #1, setsid is preferred over nohup because
  it creates a new session and process group — the BEAM spawned by the child
  is guaranteed to survive the parent exiting, regardless of how the parent's
  shell is configured.
  """

  @spec spawn_detached(Path.t(), [{charlist(), charlist()}]) ::
          {:ok, integer()} | {:error, term()}
  def spawn_detached(binary_path, env) do
    setsid = System.find_executable("setsid") || "/usr/bin/setsid"

    port =
      Port.open(
        {:spawn_executable, setsid},
        [
          :binary,
          :exit_status,
          :hide,
          args: [binary_path, "serve"],
          env: env
        ]
      )

    case Port.info(port, :os_pid) do
      {:os_pid, os_pid} ->
        Port.close(port)
        {:ok, os_pid}

      nil ->
        {:error, :no_os_pid}
    end
  rescue
    e -> {:error, Exception.message(e)}
  end
end
```

defmodule Glorbo.CLI.Lifecycle.Up do
  alias Glorbo.CLI.Lifecycle.{Pidfile, Daemon}
  alias Glorbo.Company.AuditLog

  @switches [help: :boolean]

  @spec run([String.t()]) :: Glorbo.CLI.result()
  def run(argv) do
    {opts, _, _} = OptionParser.parse(argv, strict: @switches)
    if opts[:help], do: help(), else: do_run()
  end

  defp do_run do
    case Pidfile.status() do
      :running ->
        pid = Pidfile.read!()
        {:up, 2, "glorbo is already running (pid=#{pid}). Run `glorbo down` first.\n"}

      :stale ->
        Pidfile.rm()
        _start()

      :stopped ->
        _start()
    end
  end

  defp _start do
    with {:ok, cookie} <- Glorbo.Config.erl_cookie(),
         binary <- locate_binary(),
         env = [{~c"RELEASE_COOKIE", String.to_charlist(cookie)}],
         :ok <- emit_audit("cli.up.start", %{}),
         {:ok, os_pid} <- Daemon.spawn_detached(binary, env),
         :ok <- Pidfile.write!(os_pid),
         :ok <- emit_audit("cli.up.complete", %{pid: os_pid}) do
      {:up, 0, "glorbo up (pid=#{os_pid}). Dashboard: http://127.0.0.1:4000\n"}
    else
      {:error, reason} ->
        {:up, 2, "Failed to start glorbo: #{inspect(reason)}. Run `glorbo doctor` to diagnose.\n"}
    end
  end

  defp locate_binary do
    System.get_env("__BURRITO_BIN_PATH") ||
      System.get_env("GLORBO_BINARY_PATH") ||   # test-only override
      raise "Cannot locate Glorbo binary; set __BURRITO_BIN_PATH (Burrito) or GLORBO_BINARY_PATH (test)."
  end

  defp emit_audit(action, detail) do
    AuditLog.append(%{company: "_system", actor: "cli", action: action, detail: detail})
    :ok
  rescue
    # Audit failure MUST NOT block lifecycle (CLAUDE.md audit is append-only,
    # but missing audit infra in early-boot contexts is tolerated).
    _ -> :ok
  end

  defp help do
    {:up, 0, """
    glorbo up — start glorbo in the background

    Writes pidfile to ~/.glorbo/run/glorbo.pid (mode 0600).
    Exports RELEASE_COOKIE for Erlang distribution.

    USAGE
      glorbo up [--help]
    """}
  end
end

defmodule Glorbo.CLI.Logs do
  @moduledoc "D-14 / D-15 — tail audit JSONL or agent stdout.log."

  @switches [lines: :integer, follow: :boolean, help: :boolean]

  @spec run([String.t()]) :: Glorbo.CLI.result()
  def run(argv) do
    {opts, positional, _} = OptionParser.parse(argv, strict: @switches)
    cond do
      opts[:help] -> help()
      positional == [] -> usage()
      true -> do_run(positional, opts)
    end
  end

  defp do_run([company], opts), do: tail_audit(company, opts)
  defp do_run([company, agent], opts), do: tail_stdout(company, agent, opts)
  defp do_run(_, _), do: usage()

  defp tail_audit(company, opts) do
    base = Glorbo.Filesystem.Hierarchy.default_root() |> override_with_glorbo_home()
    path = audit_path(base, company)

    case File.exists?(path) do
      true ->
        lines = opts[:lines] || 50
        backfill = read_last_lines(path, lines) |> Enum.map(&format_audit_line/1)
        Enum.each(backfill, &IO.puts/1)
        if opts[:follow], do: follow(path, :audit), else: {:logs, 0, ""}

      false ->
        {:logs, 1, "No audit log found at #{path}. Check company name.\n"}
    end
  end

  defp tail_stdout(company, agent, opts) do
    base = Glorbo.Filesystem.Hierarchy.default_root() |> override_with_glorbo_home()
    path = Path.join([base, "companies", company, "agents", agent, "stdout.log"])

    case File.exists?(path) do
      true ->
        lines = opts[:lines] || 50
        read_last_lines(path, lines) |> Enum.each(&IO.puts/1)
        if opts[:follow], do: follow(path, :stdout), else: {:logs, 0, ""}

      false ->
        {:logs, 1, "No stdout log found at #{path}. Check company/agent names.\n"}
    end
  end

  defp audit_path(base, company) do
    month = DateTime.utc_now() |> Calendar.strftime("%Y-%m")
    Path.join([base, "companies", company, "audit", "#{month}.jsonl"])
  end

  defp read_last_lines(path, n) do
    path |> File.stream!() |> Enum.take(-n)
  end

  defp format_audit_line(raw) do
    case Jason.decode(String.trim(raw)) do
      {:ok, %{"ts" => ts, "actor" => a, "action" => act, "target" => t} = m} ->
        detail = Jason.encode!(Map.get(m, "detail", %{}))
        "#{ts}  #{a}  #{act}  #{t || "-"}  #{detail}"

      _ ->
        raw  # couldn't parse → emit raw
    end
  end

  defp follow(path, kind) do
    if inotify_available?() do
      follow_inotify(path, kind)
    else
      IO.puts(:stderr, "inotifywait not available; falling back to 1s poll")
      follow_poll(path, kind)
    end
  end

  defp inotify_available?, do: System.find_executable("inotifywait") != nil

  defp follow_inotify(path, kind) do
    {:ok, watcher} = FileSystem.start_link(dirs: [Path.dirname(path)])
    FileSystem.subscribe(watcher)
    listen_loop(path, kind, File.stat!(path).size)
  end

  defp listen_loop(path, kind, last_size) do
    receive do
      {:file_event, _pid, {^path, events}} ->
        cond do
          :modified in events ->
            new_bytes = read_incremental(path, last_size)
            IO.write(if kind == :audit, do: format_audit_chunk(new_bytes), else: new_bytes)
            listen_loop(path, kind, File.stat!(path).size)

          :closed in events ->
            # month rollover possible — re-resolve
            new_path = audit_path_for_follow(path)
            listen_loop(new_path, kind, 0)

          true ->
            listen_loop(path, kind, last_size)
        end

      {:file_event, _pid, :stop} ->
        {:logs, 0, ""}
    after
      60_000 -> listen_loop(path, kind, last_size)
    end
  end

  # ... follow_poll and helpers ...

  defp read_incremental(path, from_byte) do
    {:ok, io} = File.open(path, [:read, :binary])
    :file.position(io, from_byte)
    data = IO.read(io, :eof) |> to_string()
    File.close(io)
    data
  end

  defp format_audit_chunk(bytes) do
    bytes
    |> String.split("\n", trim: true)
    |> Enum.map(&format_audit_line/1)
    |> Enum.join("\n")
    |> Kernel.<>("\n")
  end

  defp audit_path_for_follow(old_path) do
    [base, _rest] = String.split(old_path, "/audit/", parts: 2)
    month = DateTime.utc_now() |> Calendar.strftime("%Y-%m")
    "#{base}/audit/#{month}.jsonl"
  end

  defp override_with_glorbo_home(default), do: System.get_env("GLORBO_HOME") || default

  defp usage, do: {:logs, 1, "Usage: glorbo logs <company> [agent] [--lines N] [--follow]\n"}
  defp help, do: {:logs, 0, "glorbo logs — tail audit JSONL or agent stdout.log.\nUSAGE\n  glorbo logs <company> [agent] [--lines N] [--follow]\n\nDefaults: 50-line backfill. Month rollover auto-resumed when --follow.\n"}
end
