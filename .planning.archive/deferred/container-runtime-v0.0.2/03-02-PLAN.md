       defmodule Glorbo.Budget.Ledger do
         import Ecto.Query, warn: false
         alias Glorbo.{Budget, Repo}

         @spec upsert_usage(map()) :: :ok | {:error, term()}
         def upsert_usage(%{agent_slug: _, year_month: _, prompt_tokens: pt, completion_tokens: ct, cost_usd_cents: cents} = usage)
             when is_integer(pt) and is_integer(ct) and is_integer(cents) and pt >= 0 and ct >= 0 and cents >= 0 do
           case Budget.changeset(%Budget{}, usage) do
             %{valid?: true} = cs ->
               struct = Ecto.Changeset.apply_changes(cs)
               update_q =
                 from b in Budget,
                   update: [
                     inc: [
                       prompt_tokens: ^usage.prompt_tokens,
                       completion_tokens: ^usage.completion_tokens,
                       cost_usd_cents: ^usage.cost_usd_cents
                     ]
                   ]
               Repo.insert!(struct,
                 on_conflict: update_q,
                 conflict_target: [:agent_slug, :year_month]
               )
               :ok
             cs -> {:error, cs}
           end
         end

         def upsert_usage(usage), do: {:error, {:invalid_input, usage}}

         @spec current_month_usage(String.t(), String.t()) :: Budget.t() | nil
         def current_month_usage(agent_slug, year_month) do
           Repo.get_by(Budget, agent_slug: agent_slug, year_month: year_month)
         end
       end
