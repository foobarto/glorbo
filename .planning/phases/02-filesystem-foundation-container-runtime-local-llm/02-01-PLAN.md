    defmodule Glorbo.Filesystem.Hierarchy do
      @moduledoc "Materialises ~/.glorbo/ per DESIGN.md §3. Idempotent (D-19)."

      @dirs ~w(bin models containers/glorbo-runtime containers/cache companies runtime/sockets logs/containers)
      @files [{"config.md", ""}, {"logs/glorbo.log", ""}]

      @spec ensure!(Path.t()) :: :ok
      def ensure!(base) do
        Enum.each(@dirs, fn d -> File.mkdir_p!(Path.join(base, d)) end)
        Enum.each(@files, fn {path, default} ->
          full = Path.join(base, path)
          unless File.exists?(full), do: File.write!(full, default)
        end)
        File.chmod!(Path.join(base, "runtime/sockets"), 0o700)
        :ok
      end

      @spec default_root() :: Path.t()
      def default_root, do: Path.expand("~/.glorbo")
    end

    Step B — Create `lib/glorbo/filesystem/reindex.ex` modeled on RESEARCH §Pattern 3 but with the result map shape from Test 5:
    ```elixir
    defmodule Glorbo.Filesystem.Reindex do
      require Logger
      alias Glorbo.{Repo, Company, Agent, AuditEvent}
      alias Glorbo.Filesystem.{Frontmatter, ReindexState}
      import Ecto.Query

      @spec run(keyword()) :: {:ok, %{indexed: integer, skipped: integer, deleted: integer}}
      def run(opts \\ []) do
        base = Keyword.get(opts, :base, Path.expand("~/.glorbo"))
        companies_dir = Path.join(base, "companies")

        if not File.dir?(companies_dir) do
          {:ok, %{indexed: 0, skipped: 0, deleted: 0}}
        else
          files = Path.wildcard(Path.join(companies_dir, "**/*.md"))

          {indexed, skipped} =
            Enum.reduce(files, {0, 0}, fn path, {idx, skp} ->
              case process_file(path) do
                :indexed -> {idx + 1, skp}
                :unchanged -> {idx, skp}
                {:skip, reason} ->
                  Logger.warning("reindex skipped #{path}: #{inspect(reason)}")
                  {idx, skp + 1}
              end
            end)

          # D-26: delete vanished rows
          seen = MapSet.new(files)
          deleted =
            Repo.all(from r in ReindexState, select: r.file_path)
            |> Enum.filter(&(not MapSet.member?(seen, &1)))
            |> Enum.map(fn fp ->
              Repo.delete_all(from r in ReindexState, where: r.file_path == ^fp)
              cleanup_domain_row(fp)
              fp
            end)
            |> length()

          {:ok, %{indexed: indexed, skipped: skipped, deleted: deleted}}
        end
      end

      defp process_file(path) do
        content = File.read!(path)
        digest = :crypto.hash(:md5, content) |> Base.encode16(case: :lower)

        case Repo.get(ReindexState, path) do
          %{md5: ^digest} -> :unchanged
          _ ->
            case Frontmatter.parse(content) do
              {:ok, meta, _body} ->
                upsert_domain_row(path, meta)
                stat = File.stat!(path, time: :universal)
                mtime = NaiveDateTime.from_erl!(stat.mtime)
                Repo.insert!(
                  %ReindexState{file_path: path, md5: digest, size: stat.size, mtime: mtime},
                  on_conflict: {:replace, [:md5, :size, :mtime, :updated_at]},
                  conflict_target: :file_path
                )
                :indexed
              {:error, reason} -> {:skip, reason}
            end
        end
      end

      defp upsert_domain_row(path, meta) do
        cond do
          String.ends_with?(path, "/company.md") ->
            name = meta["name"] || infer_company_name(path)
            mission = meta["mission"]
            Repo.insert!(
              %Company{name: name, mission: mission, file_path: path},
              on_conflict: {:replace, [:name, :mission, :updated_at]},
              conflict_target: :file_path
            )

          String.match?(path, ~r{/agents/[^/]+/agent\.md$}) ->
            name = meta["name"] || infer_agent_name(path)
            role = meta["role"]
            provider = meta["provider"]
            model = meta["model"]

    defmodule Glorbo.Company.AuditLog do
      @moduledoc """
      Append-only audit sink. Writes JSONL to audit/YYYY-MM.jsonl under the company dir
      (or audit/_system/ for orchestrator events) AND mirrors to the audit_events table.

      LOAD-BEARING (CLAUDE.md): exports ONLY append/2. No update/delete/edit.
      Negative test: test/glorbo/stubs_test.exs.

      JSONL is the source of truth (FS-05). SQLite mirror is derived. On mirror failure,
      log + continue — JSONL write must not be rolled back.
      """
      use GenServer
      require Logger
      alias Glorbo.{Repo, AuditEvent}

      @spec start_link(keyword()) :: GenServer.on_start()
      def start_link(opts) do
        name = Keyword.get(opts, :name, __MODULE__)
        GenServer.start_link(__MODULE__, opts, name: name)
      end

      @spec append(GenServer.server() | atom(), map()) :: :ok
      def append(server \\ __MODULE__, %{} = entry) do
        GenServer.call(server, {:append, entry})
      end

      @impl GenServer
      def init(opts) do
        base = Keyword.get(opts, :base, Path.expand("~/.glorbo"))
        {:ok, %{base: base}}
      end

      @impl GenServer
      def handle_call({:append, entry}, _from, state) do
        ts = entry[:ts] || DateTime.utc_now()
        ts_iso = ts |> normalize_dt() |> DateTime.to_iso8601()
        company = to_string(entry[:company] || "_system")
        actor = to_string(entry[:actor] || "system")
        action = to_string(entry[:action] || "unknown")
        target = entry[:target]
        detail_map = Map.drop(entry, [:ts, :company, :actor, :action, :target])

        record = %{
          ts: ts_iso,
          actor: actor,
          action: action,
          target: target,
          detail: detail_map
        }

        json = Jason.encode!(record) <> "\n"
        path = jsonl_path(state.base, company, ts)
        File.mkdir_p!(Path.dirname(path))
        :ok = File.write!(path, json, [:append, :sync])

        # Mirror to SQLite — derived; failure must not roll back JSONL
        try do
          {:ok, _} =
            Repo.insert(%AuditEvent{
              company: company, actor: actor, action: action, target: target,
              detail: Jason.encode!(detail_map),
              ts: DateTime.truncate(normalize_dt(ts), :second)
            })
        rescue
          e -> Logger.error("audit_events mirror failed: #{Exception.message(e)} (JSONL still written)")
        end

        {:reply, :ok, state}
      end

      defp normalize_dt(%DateTime{} = dt), do: dt
      defp normalize_dt(_), do: DateTime.utc_now()

      defp jsonl_path(base, company, ts) do
        month = ts |> normalize_dt() |> DateTime.to_date() |> Date.to_string() |> String.slice(0, 7)
        if company == "_system" do
          Path.join([base, "audit", "_system", "#{month}.jsonl"])
        else
          Path.join([base, "companies", company, "audit", "#{month}.jsonl"])
        end
      end
    end
    ```
