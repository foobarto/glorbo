<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → LiveSocket → LiveView `handle_event` | User inputs (`task_path`, `agent`, `reason`) cross into Elixir; LV must pass them through `GlorboWeb.Actions` which re-validates (defense in depth). |
| Disk → `parse_file` → render | `agent.md` / `task.md` content may come from agent writes; `Phoenix.HTML` auto-escapes all interpolations but `{raw ...}` is forbidden on untrusted fields. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-09 | Tampering | Approval queue handle_event passes attacker-controlled `task_path` | mitigate | Pass task_path to `GlorboWeb.Actions.set_approval/4` which validates path starts with `projects/`, no `..`, ends with `.md`. LV never opens the file itself. |
| T-04-10 | XSS | Task title / agent role / company name rendered in cards | mitigate | HEEx default `{@foo}` auto-escapes. No `{raw ...}` on any field sourced from filesystem. Regression test: insert `<script>` into task title, assert rendered HTML contains `&lt;script&gt;`. |
| T-04-11 | Info disclosure | Company-scoped LV receives events for other companies (04-RESEARCH Pitfall 7) | mitigate | Every `handle_info({:file_event, rel, _}, socket)` filters `rel` against `socket.assigns.company`; tests assert cross-company event is dropped (no render change). |
| T-04-12 | DoS | StdoutStreamer leaks processes on rapid agent-page navigation (04-RESEARCH Pitfall 4) | mitigate | AgentLive `mount/3` does `Process.monitor(streamer_pid)` and `terminate/2` calls `GlorboWeb.StdoutStreamer.stop/1`. DynamicSupervisor restarts dead streamers independently of LV. |
</threat_model>

    defmodule GlorboWeb.Components.BudgetRing do
      use Phoenix.Component

      @circumference 100.53

      attr :used, :float, default: 0.0
      attr :cap, :any, default: nil
      attr :size, :integer, default: 40

      def budget_ring(assigns) do
        assigns = assigns |> assign(:ratio, ratio(assigns)) |> assign(:color, color(ratio(assigns)))
        ~H"""
        <svg class="gl-budget-ring" viewBox="0 0 36 36" width={@size} height={@size}>
          <circle cx="18" cy="18" r="16" stroke="var(--gl-surface-raised)" stroke-width="3" fill="none" />
          <circle :if={@cap} cx="18" cy="18" r="16"
                  stroke={@color} stroke-width="3" fill="none"
                  stroke-dasharray={"#{@ratio * @circumference} #{@circumference}"}
                  transform="rotate(-90 18 18)" />
          <text x="18" y="20" text-anchor="middle" font-size="8" fill="var(--gl-fg)">
            {center_text(assigns)}
          </text>
        </svg>
        """
      end

      defp ratio(%{cap: nil}), do: 0.0
      defp ratio(%{used: u, cap: c}) when is_number(u) and is_number(c) and c > 0, do: min(u / c, 1.0)
      defp ratio(_), do: 0.0

      defp color(r) when r >= 0.9, do: "var(--gl-danger)"
      defp color(r) when r >= 0.5, do: "var(--gl-warning)"
      defp color(_),               do: "var(--gl-success)"

      defp center_text(%{cap: nil, used: u}), do: "$#{:erlang.float_to_binary(u * 1.0, decimals: 2)}"
      defp center_text(%{used: u, cap: c, size: s}) when s < 60,
        do: "$#{:erlang.float_to_binary(u * 1.0, decimals: 0)}"
      defp center_text(%{used: u, cap: c}),
        do: "$#{:erlang.float_to_binary(u * 1.0, decimals: 2)}/#{:erlang.float_to_binary(c * 1.0, decimals: 0)}"
    end

      use GlorboWeb, :live_view
      alias GlorboWeb.Components.{BudgetRing, StdoutTail}

      @impl true
      def mount(%{"company" => co, "agent" => ag}, _session, socket) do
        base = Application.get_env(:glorbo, :glorbo_base, Path.expand("~/.glorbo"))
        ag_dir = Path.join([base, "companies", co, "agents", ag])

        if File.dir?(ag_dir) do
          agent_info = load_agent(base, co, ag)

          socket =
            socket
            |> assign(:page_title, "#{agent_info.name} — #{co} — Glorbo")
            |> assign(:company_slug, co)
            |> assign(:agent_slug, ag)
            |> assign(:agent, agent_info)
            |> assign(:streamer_pid, nil)
            |> stream(:stdout, [], limit: -1000)

          if connected?(socket) do
            Phoenix.PubSub.subscribe(Glorbo.PubSub, "company:#{co}:agents:#{ag}:stdout")
            Phoenix.PubSub.subscribe(Glorbo.PubSub, "company:#{co}:agents:#{ag}:wake")

            case GlorboWeb.StdoutStreamer.start(co, ag, base: base) do
              {:ok, pid} ->
                Process.monitor(pid)
                {:ok, assign(socket, :streamer_pid, pid)}
              _ ->
                {:ok, socket}
            end
          else
            {:ok, socket}
          end
        else
          {:ok,
           socket
           |> put_flash(:error, "Agent \"#{ag}\" not found in #{co}.")
           |> push_navigate(to: ~p"/companies/#{co}")}
        end
      end

      @impl true
      def handle_info({:stdout_line, _co, _ag, %{id: id, body: body}}, socket) do
        {:noreply, stream_insert(socket, :stdout, %{id: id, body: body}, at: -1, limit: -1000)}
      end

      def handle_info({:file_event, _rel, _events}, socket), do: {:noreply, socket}

      def handle_info({:DOWN, _ref, :process, pid, _reason}, %{assigns: %{streamer_pid: pid}} = socket) do
        base = Application.get_env(:glorbo, :glorbo_base, Path.expand("~/.glorbo"))
        case GlorboWeb.StdoutStreamer.start(socket.assigns.company_slug, socket.assigns.agent_slug, base: base) do
          {:ok, new_pid} ->
            Process.monitor(new_pid)
            {:noreply, assign(socket, :streamer_pid, new_pid)}
          _ -> {:noreply, assign(socket, :streamer_pid, nil)}
        end
      end

      def handle_info(_, socket), do: {:noreply, socket}

      @impl true
      def handle_event("wake", %{"reason" => reason}, socket) do

          {:error, err} -> {:noreply, put_flash(socket, :error, "Wake failed: #{inspect(err)}")}
        end
      end

      @impl true
      def terminate(_reason, socket) do
        if pid = socket.assigns[:streamer_pid], do: GlorboWeb.StdoutStreamer.stop(pid)
        :ok
      end

            <h1 class="gl-heading gl-heading--display">{@agent.name}</h1>

      use GlorboWeb, :live_view

      end

      @impl true
      def handle_info({:file_event, _rel, _events}, socket) do
        {:noreply, assign(socket, :sentinels, load_sentinels(socket.assigns.base, socket.assigns.company_slug))}
      end

      def handle_info(_, socket), do: {:noreply, socket}

      @impl true
      def handle_event("approve", %{"task_path" => tp}, socket) do
        case GlorboWeb.Actions.set_approval(socket.assigns.company_slug, tp, :approved, base: socket.assigns.base) do
          :ok -> {:noreply, put_flash(socket, :info, "Approved.")}
          {:error, err} -> {:noreply, put_flash(socket, :error, "Approve failed: #{inspect(err)}")}
        end
      end

      def handle_event("deny", %{"task_path" => tp}, socket) do
        case GlorboWeb.Actions.set_approval(socket.assigns.company_slug, tp, :denied, base: socket.assigns.base) do
          :ok -> {:noreply, put_flash(socket, :info, "Denied. Task moved to history.")}
          {:error, err} -> {:noreply, put_flash(socket, :error, "Deny failed: #{inspect(err)}")}
        end
      end

      @impl true
      def render(assigns) do
        ~H"""
        <section class="gl-view gl-approvals">
          <header class="gl-view__header">
            <h1 class="gl-heading gl-heading--display">
              Approvals <span class="gl-muted">({length(@sentinels)} pending)</span>
            </h1>
          </header>

          <div :if={@sentinels == []} class="gl-empty">
            <p>No approvals pending.</p>
            <p class="gl-muted">Tasks with <code>requires_approval: director</code> in frontmatter will appear here.</p>
          </div>

          <div :if={@sentinels != []} class="gl-approvals__list">
            <ApprovalCard.approval_card :for={s <- @sentinels} sentinel={s} />
          </div>
        </section>
        """
      end

      defp load_sentinels(base, co) do
        agents_dir = Path.join([base, "companies", co, "agents"])

        case File.ls(agents_dir) do
          {:ok, agents} ->
            Enum.flat_map(agents, fn ag ->
              state_dir = Path.join([agents_dir, ag, "state"])
              case File.ls(state_dir) do
                {:ok, files} ->
                  files
                  |> Enum.filter(&String.starts_with?(&1, "awaiting-approval-"))
                  |> Enum.map(&build_sentinel(base, co, ag, &1))
                  |> Enum.filter(&(&1 != nil))
                _ -> []
              end
            end)
          _ -> []
        end
      end

      defp build_sentinel(base, co, agent, filename) do
        task_id =
          filename
          |> String.replace_prefix("awaiting-approval-", "")
          |> String.replace_suffix(".md", "")

        task_path = find_task_path(base, co, task_id)

        case task_path && Glorbo.TaskDefinition.parse_file(Path.join([base, "companies", co, task_path]), base: base, company: co) do
          {:ok, task} ->
            sentinel_path = Path.join([base, "companies", co, "agents", agent, "state", filename])
            requested_at = case File.stat(sentinel_path, time: :posix) do
              {:ok, %File.Stat{mtime: m}} -> DateTime.from_unix!(m) |> DateTime.to_iso8601()
              _ -> ""
            end

            %{
              task_id: task_id,
              task_path: task_path,
              title: task.title || task_id,
              requesting_agent: agent,
              requested_at: requested_at
            }
          _ -> nil
        end
      end

      defp find_task_path(base, co, task_id) do
        projects_dir = Path.join([base, "companies", co, "projects"])
        case File.ls(projects_dir) do
          {:ok, projects} ->
            Enum.find_value(projects, fn p ->
              candidate = Path.join([projects_dir, p, "tasks", "#{task_id}.md"])
              if File.exists?(candidate), do: "projects/#{p}/tasks/#{task_id}.md"
            end)
          _ -> nil
        end
      end
    end
