<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Channel markdown content → earmark → sanitizer → HEEx | User-typed (Director or agent) body crosses into HTML; must sanitize before render. |
| `Application.get_env(:glorbo, :dashboard_token)` → `Plug.Conn` query_params | Secret string compared to client-supplied param; timing attack possible if not secure_compare. |
| `audit/YYYY-MM.jsonl` tail → AuditLive render | Audit payloads contain filesystem paths, agent names — paths could be adversarial if an agent wrote to audit (only AuditLog writes, but detail maps contain agent-supplied strings). |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-13 | XSS | Channel message body contains `<script>` / `javascript:` URLs | mitigate | `GlorboWeb.Markdown.render/2` pipeline: (1) pre-pass `@mention` substitution into sanitized anchor tags, (2) `Earmark.as_html!(..., compact_output: true, smartypants: false)`, (3) `HtmlSanitizeEx.markdown_html/1`. Regression test asserts `<script>` in raw body produces NO `<script>` in output. |
| T-04-14 | Spoofing | Dashboard token timing attack | mitigate | `Plug.Crypto.secure_compare/2` for token comparison (constant-time). Never include the token in error messages or logs. |
| T-04-15 | Information disclosure | AuditLive renders audit payloads that include filesystem paths / detail strings | accept | Audit log is the authoritative record; Director is trusted operator. Paths are scoped to `~/.glorbo/`. Not a threat in single-operator trust model. |
| T-04-16 | DoS | AuditLive loads entire month JSONL into memory on mount | mitigate | Stream file + `Enum.take(-500)` tail; `Phoenix.LiveView.stream/4` with `limit: -500`; "Load older" prepends next 500 batch — bounded memory. |
| T-04-17 | Tampering | Sidebar renders company names read from filesystem | mitigate | HEEx auto-escapes interpolated strings; rename by agent can't inject HTML. |
| T-04-18 | XSS | Mention pre-pass regex allows HTML in replacement string | mitigate | Mention regex captures only `[a-z0-9-]+`; replacement is a fixed template with captured slug HTML-escaped via `Phoenix.HTML.Safe`. Test with `@<script>` input. |
</threat_model>

    ```elixir
    defmodule GlorboWeb.Markdown do
      @moduledoc """
      Channel message markdown pipeline:

        1. Mention pre-pass: `@slug` → `<a class="gl-mention" href="/companies/:co/agents/:slug">@slug</a>`
        2. Earmark GFM rendering
        3. `HtmlSanitizeEx.markdown_html/1` — strips `<script>`, `<iframe>`, `javascript:` URLs, etc.

      Output is safe HTML iodata wrapped in `{:safe, _}`; pass through `Phoenix.HTML.raw/1`
      at render site.
      """

      @mention_re ~r/@([a-z0-9-]+)/

      @spec render(binary(), keyword()) :: {:safe, iodata()}
      def render(body, opts \\ []) when is_binary(body) do
        company = Keyword.get(opts, :company, "")
        safe_company = Phoenix.HTML.html_escape(company) |> Phoenix.HTML.safe_to_string()

        body
        |> mention_pre_pass(safe_company)
        |> earmark_render()
        |> sanitize()
        |> Phoenix.HTML.raw()
      end

      defp mention_pre_pass(body, safe_company) do
        Regex.replace(@mention_re, body, fn _, slug ->
          safe_slug = Phoenix.HTML.html_escape(slug) |> Phoenix.HTML.safe_to_string()
          ~s(<a class="gl-mention" href="/companies/#{safe_company}/agents/#{safe_slug}">@#{safe_slug}</a>)
        end)
      end

      defp earmark_render(body) do
        Earmark.as_html!(body, %Earmark.Options{compact_output: true, smartypants: false, gfm: true})
      end

      defp sanitize(html), do: HtmlSanitizeEx.markdown_html(html)
    end
    ```

    ```elixir
    defmodule GlorboWeb.ChannelLive do
      use GlorboWeb, :live_view
      alias GlorboWeb.Components.ChannelMessage

      @message_re ~r/^## (?<ts>[^|]+?)\s*\|\s*(?<author>.+?)\s*\n(?<body>.*?)(?=\n## |\z)/ms

      @impl true
      def mount(%{"company" => co, "channel" => ch}, _session, socket) do
        base = Application.get_env(:glorbo, :glorbo_base, Path.expand("~/.glorbo"))
        path = Path.join([base, "companies", co, "channels", "#{ch}.md"])

        if File.exists?(path) do
          if connected?(socket), do: Phoenix.PubSub.subscribe(Glorbo.PubSub, "company:#{co}:channels:#{ch}")

          {:ok,
           socket
           |> assign(:page_title, "#{ch} — #{co} — Glorbo")
           |> assign(:company_slug, co)
           |> assign(:current_company, co)
           |> assign(:channel, ch)
           |> assign(:base, base)
           |> assign(:messages, load_messages(path, co))
           |> assign(:compose_body, "")}
        else
          {:ok,
           socket
           |> put_flash(:error, "Channel \"#{ch}\" not found in #{co}.")
           |> push_navigate(to: ~p"/companies/#{co}")}
        end
      end

      @impl true
      def handle_info({:file_event, _rel, _events}, socket) do
        path = Path.join([socket.assigns.base, "companies", socket.assigns.company_slug,
                          "channels", "#{socket.assigns.channel}.md"])
        {:noreply, assign(socket, :messages, load_messages(path, socket.assigns.company_slug))}
      end

      def handle_info(_, socket), do: {:noreply, socket}

      @impl true
      def handle_event("post", %{"body" => body}, socket) do
        case GlorboWeb.Actions.post_message(
               socket.assigns.company_slug,
               socket.assigns.channel,
               String.trim(body),
               base: socket.assigns.base
             ) do
          :ok -> {:noreply, assign(socket, :compose_body, "")}
          {:error, :empty_body} -> {:noreply, put_flash(socket, :error, "Message is empty.")}
          {:error, :body_too_large} -> {:noreply, put_flash(socket, :error, "Message exceeds 10 KB.")}
          {:error, reason} -> {:noreply, put_flash(socket, :error, "Failed to post: #{inspect(reason)}. Message not sent.")}
        end
      end

      @impl true
      def render(assigns) do
        ~H"""
        <section class="gl-view gl-channel">
          <header class="gl-view__header">
            <h1 class="gl-heading gl-heading--display">#{@channel}</h1>
          </header>

          <div :if={@messages == []} class="gl-empty">
            <p>No messages in #{@channel} yet.</p>
          </div>

          <div :if={@messages != []} class="gl-channel__messages">
            <ChannelMessage.channel_message :for={m <- @messages} message={m} />
          </div>

          <form phx-submit="post" class="gl-compose">
            <textarea name="body" maxlength="10240" class="gl-input"
                      placeholder={"Message ##{@channel} as Director…"}>{@compose_body}</textarea>
            <button type="submit" class="gl-btn gl-btn--primary">Send</button>
          </form>
        </section>
        """
      end

      defp load_messages(path, company) do
        case File.read(path) do
          {:ok, content} -> parse_messages(content, company)
          _ -> []
        end
      end

      defp parse_messages(content, company) do
        Regex.scan(@message_re, content, capture: :all_names)
        |> Enum.map(fn [author, body, ts] ->
          %{
            author: String.trim(author),
            timestamp: String.trim(ts),
            body_html: GlorboWeb.Markdown.render(String.trim(body), company: company)
          }
        end)
      end
    end
    ```

    ```elixir
    defmodule GlorboWeb.Plugs.DashboardToken do
      @moduledoc """
      Optional bearer-token gate (D-06).

      When `Application.get_env(:glorbo, :dashboard_token)` is nil (default),
      this plug is a no-op — the loopback bind in `config/runtime.exs` is the
      only trust boundary.

      When set, the request must carry `?token=<value>` matching the configured
      string via `Plug.Crypto.secure_compare/2` (constant-time). Mismatch → 401.
      """
      import Plug.Conn

      def init(opts), do: opts

      def call(conn, _opts) do
        case Application.get_env(:glorbo, :dashboard_token) do
          nil ->
            conn
          "" ->
            conn
          expected when is_binary(expected) ->
            conn = fetch_query_params(conn)
            supplied = conn.query_params["token"] || ""

            if Plug.Crypto.secure_compare(expected, supplied) do
              conn
            else
              conn
              |> send_resp(401, "unauthorized")
              |> halt()
            end
        end
      end
    end
    ```

    ```elixir
      alias GlorboWeb.Components.AuditEntry

      @poll_ms 1_000

      @impl true
      def mount(%{"company" => co}, _session, socket) do

        ym = current_year_month()
        path = Path.join([base, "companies", co, "audit", "#{ym}.jsonl"])

         |> assign(:page_title, "Audit — #{co} — Glorbo")

         |> assign(:path, path)
         |> assign(:actor_filter, "")
         |> assign(:action_filter, "")
         |> assign(:offset, offset)
         |> assign(:expanded, MapSet.new())
         |> assign(:beginning, offset == 0)
         |> assign(:entries, entries)}
      end

      @impl true
      def handle_info(:poll, socket) do
        Process.send_after(self(), :poll, @poll_ms)
        {new_entries, _} = load_tail(socket.assigns.path, 500)
        {:noreply, assign(socket, :entries, new_entries)}
      end

      @impl true
      def handle_event("filter", %{"actor" => a, "action" => act}, socket) do
        {:noreply, assign(socket, actor_filter: a, action_filter: act)}
      end

      def handle_event("toggle", %{"id" => id}, socket) do
        exp = if MapSet.member?(socket.assigns.expanded, id), do: MapSet.delete(socket.assigns.expanded, id), else: MapSet.put(socket.assigns.expanded, id)
        {:noreply, assign(socket, :expanded, exp)}
      end

      def handle_event("load_older", _, socket) do
        case load_older(socket.assigns.path, socket.assigns.offset, 500) do
          {[], _} -> {:noreply, assign(socket, :beginning, true)}
          {older, new_offset} ->
            {:noreply,
             socket
             |> assign(:entries, older ++ socket.assigns.entries)
             |> assign(:offset, new_offset)
             |> assign(:beginning, new_offset == 0)}
        end
      end

        filtered = filter_entries(assigns.entries, assigns.actor_filter, assigns.action_filter)
        assigns = assign(assigns, :filtered, filtered)

        ~H"""
        <section class="gl-view gl-audit">
          <header class="gl-view__header">

            <input type="text" name="action" value={@action_filter} class="gl-input" placeholder="Filter by action…" />
          </form>

        </section>
        """
      end

      defp current_year_month do
        d = Date.utc_today()
        "#{d.year}-#{String.pad_leading(Integer.to_string(d.month), 2, "0")}"

      defp load_tail(path, n) do
        case File.read(path) do
          {:ok, content} ->
            lines = content |> String.split("\n", trim: true)
            tail = Enum.take(lines, -n)
            entries = Enum.map(tail, &decode/1) |> Enum.filter(&(&1 != nil))

            {entries, offset}
          _ ->

            entries = Enum.map(slice, &decode/1) |> Enum.filter(&(&1 != nil))
            {entries, new_offset}
          _ -> {[], 0}
        end
      end

          {:ok, m} -> m
          _ -> nil
        end
      end

          (action_f == "" or String.contains?(to_string(e["action"] || ""), action_f))
    ```
