Sandbox.Bwrap public API:
```elixir
# lib/glorbo/sandbox/bwrap.ex
@type network_policy :: :none | :api_only | :open
@type invocation_opts :: %{
        agent_workspace: String.t(),      # absolute path to agent's workspace
        inbox_path: String.t(),
        outbox_path: String.t(),
        company_path: String.t(),          # base of this company's dir (for per-permission mounts)
        permissions: [ACLMapper.permission()],
        network_policy: network_policy(),
        cli_auth_binds: [{host_path :: String.t(), sandbox_path :: String.t()}],   # e.g. [{"~/.claude", "/host-claude"}]
        cli_env: %{optional(String.t()) => String.t()},  # from Adapter.env/2
        proxy_url: String.t() | nil,        # set when network_policy == :api_only
        timeout_seconds: pos_integer()
      }

@spec build_argv(invocation_opts()) :: [String.t()]
# Pure — composes the bwrap argv from opts. Unit-tested without running bwrap.
# Returns the FULL argv starting AFTER the bwrap binary (caller prepends `bwrap`).

@spec start(invocation_opts(), run_opts :: keyword()) :: {:ok, %{exit_status: integer(), usage_dir: String.t() | nil, stdout: binary()}} | {:error, term()}
# run_opts: :cli_binary (required — path to claude/gemini/codex), :cli_args (from Adapter.args),
#           :prompt (stdin content), :stdout_log (file for tailing), :bwrap_binary.
# Internal: Port.open via MuonTrap.Daemon; stdin-write the prompt; wait for exit; return usage dir path for Dispatch to tail.

@spec default_binary() :: String.t()
# Returns `System.find_executable("bwrap")` or raises.
```
