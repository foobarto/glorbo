    `containers/glorbo-runtime/worker/routes.py`:
    ```python
    """HTTP routes — /run (POST), /cancel (POST). D-36, D-42."""
    import asyncio
    from typing import Optional, List
    from fastapi import APIRouter, HTTPException
    from pydantic import BaseModel
    from worker.dispatch import run_task
    from worker.context import load_task_context

    router = APIRouter()

    # Track live tasks for cancellation (D-42).
    _live_tasks: dict[str, asyncio.Task] = {}

    class RunRequest(BaseModel):
        task_path: str
        provider: str
        model: str
        api_key: Optional[str] = None
        skills: List[str] = []
        timeout_seconds: Optional[int] = 300
        request_id: str

    class RunResponse(BaseModel):
        ok: bool
        result: Optional[dict] = None
        error: Optional[str] = None

    @router.post("/run", response_model=RunResponse)
    async def run(req: RunRequest) -> RunResponse:
        ctx = load_task_context(req.task_path, req.skills)
        task = asyncio.create_task(
            run_task(ctx, req.provider, req.model, req.api_key, req.timeout_seconds)
        )
        _live_tasks[req.request_id] = task
        try:
            result = await asyncio.wait_for(task, timeout=req.timeout_seconds)
            return RunResponse(ok=True, result=result)
        except asyncio.TimeoutError:
            return RunResponse(ok=False, error="timeout")
        except asyncio.CancelledError:
            return RunResponse(ok=False, error="cancelled")
        except Exception as e:
            return RunResponse(ok=False, error=str(e))
        finally:
            _live_tasks.pop(req.request_id, None)

    class CancelRequest(BaseModel):
        request_id: str

    @router.post("/cancel")
    async def cancel(req: CancelRequest) -> dict:
        task = _live_tasks.get(req.request_id)
        if task and not task.done():
            task.cancel()
            return {"ok": True, "cancelled": True}
        return {"ok": True, "cancelled": False, "reason": "no_live_task"}
    ```

    `containers/glorbo-runtime/worker/dispatch.py`:
    ```python
    """Unified LLM dispatch via litellm (D-40).

    NEVER call provider-native SDKs from this module. The whole point of D-40
    is one error taxonomy and one provider-string format: '{provider}/{model}'.
    """
    import litellm
    from typing import Optional

    async def run_task(ctx: dict, provider: str, model: str,
                       api_key: Optional[str], timeout_seconds: int) -> dict:
        # litellm provider-string convention
        model_str = f"{provider}/{model}" if "/" not in model else model
        kwargs = {
            "model": model_str,
            "messages": ctx["messages"],
            "timeout": timeout_seconds,
        }
        if api_key:
            kwargs["api_key"] = api_key  # D-37: per-request only, never env var

        # litellm is sync-by-default; run in executor to avoid blocking the loop.
        import asyncio
        loop = asyncio.get_running_loop()
        response = await loop.run_in_executor(None, lambda: litellm.completion(**kwargs))
        return {
            "completion": response["choices"][0]["message"]["content"],
            "usage": response.get("usage", {}),
            "model": response.get("model", model_str),
        }
    ```

    `containers/glorbo-runtime/worker/context.py`:
    ```python
    """Reads task.md + agent.md + skills/*.md from the bind-mounted /company/ tree.

    D-36: 'Filesystem is the source of truth' on both sides of the container boundary.
    Python reads markdown directly from the mounted FS — no copies passed over the wire.
    """
    import os
    from pathlib import Path
    from typing import List
    import yaml

    def load_task_context(task_path: str, skills: List[str]) -> dict:
        task_p = Path(task_path)
        if not task_p.exists():
            raise FileNotFoundError(f"task file not found: {task_path}")

        task_meta, task_body = _split_frontmatter(task_p.read_text())
        agent_md = _find_agent_md(task_p)
        agent_meta, agent_prompt = _split_frontmatter(agent_md.read_text()) if agent_md else ({}, "")
        skill_blocks = _load_skills(task_p, skills)

        messages = [
            {"role": "system", "content": agent_prompt + "\n\n" + "\n\n".join(skill_blocks)},
            {"role": "user", "content": task_body},
        ]
        return {"task_meta": task_meta, "agent_meta": agent_meta, "messages": messages}

    def _split_frontmatter(text: str):
        if text.startswith("---\n"):
            _, fm, body = text.split("---\n", 2)
            return yaml.safe_load(fm) or {}, body
        return {}, text

    def _find_agent_md(task_path: Path):
        # /company/agents/<name>/inbox/<task>.md → /company/agents/<name>/agent.md
        parts = task_path.parts
        if "agents" in parts:
            idx = parts.index("agents")
            agent_dir = Path(*parts[: idx + 2])
            candidate = agent_dir / "agent.md"
            if candidate.exists():
                return candidate
        return None

    def _load_skills(task_path: Path, skills: List[str]) -> List[str]:
        # /company/skills/<name>.md
        parts = task_path.parts
        if "company" in parts:
            idx = parts.index("company")
            skills_dir = Path(*parts[: idx + 1]) / "skills"
            return [(skills_dir / f"{s}.md").read_text() for s in skills if (skills_dir / f"{s}.md").exists()]
        return []
    ```

    ```elixir
      @moduledoc """
      Builds the podman-run argv list per RT-04. Pure — no IO. Caller passes the
      result to `System.cmd("podman", argv, ...)` or MuonTrap.

      Security flags are non-negotiable. Tests assert their presence.
      """

      @type mode :: :ephemeral | :persistent

      @spec build_argv(String.t(), String.t(), mode(), keyword()) :: [String.t()]
      def build_argv(company, agent, mode \\ :ephemeral, opts \\ []) do
        base = Keyword.get(opts, :base, Path.expand("~/.glorbo"))

        host_socket_dir = Path.join([base, "runtime", "sockets", company])

        lifecycle_flag = if mode == :ephemeral, do: "--rm", else: "-d"

        [
          "run",
          lifecycle_flag,
          "--name", "glorbo-#{company}-#{agent}",
          "--userns", "keep-id",
          "--read-only",
          "--network", "none",
          "--tmpfs", "/tmp",
          "--volume", "#{host_company_dir}:/company:Z,ro",
          "--volume", "#{host_socket_dir}:/run:Z,rw",
          "--env", "GLORBO_COMPANY=#{company}",
          "--env", "GLORBO_AGENT=#{agent}",
          image,
          "uvicorn", "worker.main:app",
          "--uds", "/run/agent.sock"
        ]
      end

      @spec runtime_image() :: String.t()
      def runtime_image, do: @runtime_image_default
    end
    ```

    ```elixir
    defmodule Glorbo.Container.WorkerClient do
      @moduledoc """
      HTTP client for the FastAPI worker over Unix domain socket (D-34).
      Uses the shared Glorbo.Finch pool with `unix_socket:` option.

      Implements retry-connect backoff per D-39: 50/100/200/500ms, ~5s total.
      """
      require Logger

      @backoff_ms [50, 100, 200, 500, 1000, 2000]   # ~5 seconds total

      @spec post_run(String.t(), String.t(), map(), keyword()) :: {:ok, map()} | {:error, term()}
      def post_run(company, agent, body, opts \\ []) do
        post(company, agent, "/run", body, opts)
      end

      @spec post_cancel(String.t(), String.t(), String.t(), keyword()) :: {:ok, map()} | {:error, term()}
      def post_cancel(company, agent, request_id, opts \\ []) do
        post(company, agent, "/cancel", %{request_id: request_id}, opts)
      end

      defp post(company, agent, path, body, opts) do
        base = Keyword.get(opts, :base, Path.expand("~/.glorbo"))
        sock = Glorbo.Container.Socket.path(base, company, agent)
        timeout = Keyword.get(opts, :receive_timeout, 300_000)
        request_fun = Keyword.get(opts, :request_fun, &default_request/3)

        try_with_backoff(@backoff_ms, fn ->
          req = Finch.build(:post, "http://localhost#{path}",
                            [{"content-type", "application/json"}],
                            Jason.encode!(body))
          request_fun.(req, sock, timeout)
        end)
      end

      defp default_request(req, sock, timeout) do
        Finch.request(req, Glorbo.Finch, unix_socket: sock, receive_timeout: timeout)
        |> normalize_response()
      end

      defp normalize_response({:ok, %Finch.Response{status: 200, body: body}}),
        do: {:ok, Jason.decode!(body)}
      defp normalize_response({:ok, %Finch.Response{status: status, body: body}}),
        do: {:error, {:http_status, status, body}}
      defp normalize_response({:error, _} = err), do: err

      defp try_with_backoff([], fun), do: fun.()
      defp try_with_backoff([wait | rest], fun) do
        case fun.() do
          {:error, %Mint.TransportError{reason: :econnrefused}} ->
            Process.sleep(wait)
            try_with_backoff(rest, fun)
          {:error, %Mint.TransportError{reason: :enoent}} ->
            Process.sleep(wait)
            try_with_backoff(rest, fun)
          result -> result
        end
      end
    end
    ```

    ```elixir
    defmodule Glorbo.ContainerManager do
      @moduledoc "Manages the Podman container lifecycle. Implemented in Phase 2 Plan 03."
      use GenServer
      require Logger
      alias Glorbo.Container.{Invocation, Socket}

      @podman "podman"

      @spec start_link(keyword()) :: GenServer.on_start()
      def start_link(opts \\ []) do
        opts = Keyword.put_new(opts, :name, __MODULE__)
        GenServer.start_link(__MODULE__, opts, name: Keyword.fetch!(opts, :name))
      end

      @spec ensure_image(String.t()) :: :ok | {:error, term()}
      def ensure_image(image) do
        GenServer.call(__MODULE__, {:ensure_image, image}, 60_000)
      end

      @spec start_container(String.t(), keyword()) :: {:ok, String.t()} | {:error, term()}
      def start_container(company, opts) do
        GenServer.call(__MODULE__, {:start_container, company, opts}, 30_000)
      end

      @spec stop_container(String.t()) :: :ok | {:error, term()}
      def stop_container(name) do
        GenServer.call(__MODULE__, {:stop_container, name}, 15_000)
      end

      @impl GenServer
      def init(_opts), do: {:ok, %{}}

      @impl GenServer
      def handle_call({:ensure_image, image}, _from, state) do
        # Idempotent: if image present locally, no-op; otherwise pull.
        case System.cmd(@podman, ["image", "exists", image], stderr_to_stdout: true) do
          {_, 0} -> {:reply, :ok, state}
          {_, _} ->
            case System.cmd(@podman, ["pull", image], stderr_to_stdout: true) do
              {_, 0} -> {:reply, :ok, state}
              {output, code} -> {:reply, {:error, {:pull_failed, code, output}}, state}
            end
        end
      end

      def handle_call({:start_container, company, opts}, _from, state) do
        agent = Keyword.fetch!(opts, :agent)
        mode = Keyword.get(opts, :mode, :ephemeral)
        base = Keyword.get(opts, :base, Path.expand("~/.glorbo"))

        Socket.ensure_dir!(base, company)
        Socket.cleanup_stale(base, company, agent)

        argv = Invocation.build_argv(company, agent, mode, base: base)

        case mode do
          :ephemeral ->
            case System.cmd(@podman, argv, stderr_to_stdout: true) do
              {output, 0} -> {:reply, {:ok, String.trim(output)}, state}
              {output, code} -> {:reply, {:error, {:run_failed, code, output}}, state}
            end
          :persistent ->
            # MuonTrap-supervised; returns a port-pid handle wrapped as opaque id.
            {:ok, _pid} = MuonTrap.Daemon.start_link(@podman, argv,
              log_output: :info, stderr_to_stdout: true)
            {:reply, {:ok, "glorbo-#{company}-#{agent}"}, state}
        end
      end

      def handle_call({:stop_container, name}, _from, state) do
        case System.cmd(@podman, ["stop", name], stderr_to_stdout: true) do
          {_, 0} -> {:reply, :ok, state}
          {output, code} -> {:reply, {:error, {:stop_failed, code, output}}, state}
        end
      end
    end
    ```
