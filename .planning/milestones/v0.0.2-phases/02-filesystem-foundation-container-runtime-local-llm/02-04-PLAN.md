    ```elixir
    defmodule Glorbo.Filesystem.Watcher do
      @moduledoc """
      Per-company inotify watcher (D-30). Debounces (100ms, D-32), routes by
      path prefix (D-33). Started under Glorbo.Company.Supervisor.

      Message flow:
        FileSystem.subscribe/1 → handle_info({:file_event, _, {path, evs}})
          → if evs ⊆ [:created, :modified, :deleted, :removed]: Process.send_after(flush, 100ms)
          → handle_info({:flush, path}) → dispatch_by_prefix(company, path)
      """
      use GenServer
      require Logger

      @debounce_ms 100
      @interesting_events [:created, :modified, :deleted, :removed]

      @spec start_link(keyword()) :: GenServer.on_start()
      def start_link(opts) do
        company = Keyword.fetch!(opts, :company)
        GenServer.start_link(__MODULE__, opts, name: via(company))
      end

      @spec via(String.t()) :: {:global, {:glorbo_watcher, String.t()}}
      def via(company), do: {:global, {:glorbo_watcher, company}}

      @impl GenServer
      def init(opts) do
        company = Keyword.fetch!(opts, :company)
        base = Keyword.get(opts, :base, Path.expand("~/.glorbo"))
        company_dir = Path.join([base, "companies", company])
        File.mkdir_p!(company_dir)

        {:ok, pid} = FileSystem.start_link(dirs: [company_dir], recursive: true)
        FileSystem.subscribe(pid)

        {:ok, %{
          company: company,
          base: base,
          dir: company_dir,
          pending: %{},
          reindex_fun: Keyword.get(opts, :reindex_fun, &Glorbo.Filesystem.Reindex.mark_dirty/2)
        }}
      end

      @impl GenServer
      def handle_info({:file_event, _pid, {path, events}}, state) do
        if Enum.any?(events, &(&1 in @interesting_events)) do
          # Cancel prior timer for the same path (coalesce bursts, D-32)
          case Map.get(state.pending, path) do
            nil -> :ok
            ref -> Process.cancel_timer(ref)
          end
          ref = Process.send_after(self(), {:flush, path}, @debounce_ms)
          {:noreply, put_in(state.pending[path], ref)}
        else
          {:noreply, state}
        end
      end

      def handle_info({:flush, path}, state) do
        dispatch_by_prefix(state.company, state.dir, path, state.reindex_fun)
        {:noreply, update_in(state.pending, &Map.delete(&1, path))}
      end

      def handle_info({:file_event, _pid, :stop}, state) do
        Logger.warning("FileWatcher for #{state.company} received :stop")
        {:stop, :normal, state}
      end

      def handle_info(_, state), do: {:noreply, state}

      # D-33: path-prefix routing
      defp dispatch_by_prefix(company, base_dir, path, reindex_fun) do
        rel = Path.relative_to(path, base_dir)
        cond do
          String.starts_with?(rel, "agents/") and String.contains?(rel, "/inbox/") ->
            Logger.debug("[watcher/#{company}] inbox event #{rel} (Phase 3 routing target)")
            :ok
          String.starts_with?(rel, "agents/") and String.contains?(rel, "/outbox/") ->
            Logger.debug("[watcher/#{company}] outbox event #{rel} (Phase 3 routing target)")
            :ok
          String.starts_with?(rel, "audit/") ->
            Logger.debug("[watcher/#{company}] audit event #{rel} (observed)")
            :ok
          String.starts_with?(rel, "channels/") ->
            Logger.debug("[watcher/#{company}] channel event #{rel} (Phase 4 target)")
            :ok
          true ->
            reindex_fun.(company, path)
        end
      end
    end
    ```

    defmodule Glorbo.Init.Orchestrator do
      @moduledoc "7-step init pipeline (D-21). Continue-on-error (D-20). One audit per step (D-24)."
      require Logger
      alias Glorbo.{Doctor}
      alias Glorbo.Company.AuditLog
      alias Glorbo.Filesystem.{Hierarchy, Reindex}
      alias Glorbo.Init.{BinaryBootstrap, ImagePull, ExampleCompany}

      @type step_result :: %{step: atom(), status: :ok | :skipped | :error, detail: String.t()}
      @type summary :: %{
        results: [step_result()],
        exit_code: 0 | 1 | 2,
        failures: [step_result()],
        next_steps: [String.t()]
      }

      @spec run(keyword()) :: {:ok | :error, summary()}
      def run(opts \\ []) do
        base = Keyword.get(opts, :base, Path.expand("~/.glorbo"))
        opts_with_base = Keyword.put(opts, :base, base)

        # Ensure hierarchy + audit log BEFORE first audit event.
        Hierarchy.ensure!(base)

        # W3: Always call start_link/1 unconditionally — let {:error, {:already_started, pid}}
        # from a rerun be handled explicitly. NO Process.whereis/1 check (race-free, simpler).
        case AuditLog.start_link(name: AuditLog, base: base, company: "_system") do
          {:ok, _pid} -> :ok
          {:error, {:already_started, _pid}} -> :ok
          {:error, other} -> raise "AuditLog start failed: #{inspect(other)}"
        end

        steps = [
          {:pre_doctor,       fn o -> step_pre_doctor(o) end},
          {:hierarchy,        fn o -> step_hierarchy(o) end},
          {:binary_bootstrap, fn o -> step_binary_bootstrap(o) end},
          {:image_pull,       fn o -> ImagePull.run(o) end},
          {:example_company,  fn o -> step_example_company(o) end},
          {:reindex,          fn o -> step_reindex(o) end},
          {:post_doctor,      fn o -> step_post_doctor(o) end}
        ]

        results =
          Enum.map(steps, fn {name, fun} ->
            print_start(name)
            result = try do
              r = fun.(opts_with_base)
              Map.put(r, :step, name)
            rescue
              e -> %{step: name, status: :error, detail: Exception.message(e)}
            end
            audit_step(result)
            print_line(result)
            result
          end)

        summary = summarize(results)
        if summary.exit_code == 1, do: {:error, summary}, else: {:ok, summary}
      end

      defp step_pre_doctor(_opts) do
        checks = Doctor.run_checks()
        code = Doctor.exit_code(checks)
        %{status: code_to_status(code), detail: "pre-doctor exit #{code}"}
      end

      defp step_hierarchy(opts) do
        Hierarchy.ensure!(Keyword.fetch!(opts, :base))
        %{status: :ok, detail: "~/.glorbo/ materialised"}
      end

      defp step_binary_bootstrap(opts) do
        if Keyword.get(opts, :skip_pull, false) do
          %{status: :skipped, detail: "--skip-pull"}
        else
          pr = BinaryBootstrap.ensure_podman(opts)
          or_ = BinaryBootstrap.ensure_ollama(opts)
          combine([pr, or_])
        end
      end

      defp step_example_company(opts) do
        if Keyword.get(opts, :example, true) do
          case ExampleCompany.scaffold!(opts) do
            :ok -> %{status: :ok, detail: "acme scaffolded"}
            :already_exists -> %{status: :skipped, detail: "acme already present"}
          end
        else
          %{status: :skipped, detail: "--no-example"}
        end
      end

      defp step_reindex(opts) do
        case Reindex.run(opts) do
          {:ok, m} -> %{status: :ok, detail: "indexed=#{m.indexed} skipped=#{m.skipped} deleted=#{m.deleted}"}
          other -> %{status: :error, detail: inspect(other)}
        end
      end

      defp step_post_doctor(_opts) do
        checks = Doctor.run_checks()
        code = Doctor.exit_code(checks)
        %{status: code_to_status(code), detail: "post-doctor exit #{code}", post_doctor_code: code}
      end

      # W4: combine/1 uses match?/2 — no broken `:_` literal, no elem_match helper.
      defp combine(results) do
        cond do
          Enum.all?(results, &match?({:ok, :system, _}, &1)) ->
            %{status: :ok, mode: :no_op, detail: "all binaries already present"}
          Enum.any?(results, &match?({:error, _}, &1)) ->
            errors = Enum.filter(results, &match?({:error, _}, &1))
            %{status: :error, errors: errors, detail: "binary bootstrap had errors: #{inspect(errors)}"}
          true ->
            %{status: :ok, mode: :mixed, detail: "bootstrapped (mixed system/downloaded)"}
        end
      end

      defp code_to_status(0), do: :ok
      defp code_to_status(2), do: :skipped
      defp code_to_status(_), do: :error

      defp audit_step(%{step: step, status: status, detail: detail}) do
        AuditLog.append(%{
          company: "_system", actor: "init", action: "init.step.#{step}",
          target: to_string(step), status: to_string(status), detail: detail
        })
      end

      defp print_start(name), do: IO.puts("⧖ #{name} ...")

      defp print_line(%{step: step, status: :ok, detail: d}), do: IO.puts("  ✓ #{step} — #{d}")
      defp print_line(%{step: step, status: :skipped, detail: d}), do: IO.puts("  ⏭ #{step} — #{d}")
      defp print_line(%{step: step, status: :error, detail: d}), do: IO.puts("  ✗ #{step} — #{d}")

      defp summarize(results) do
        failures = Enum.filter(results, &(&1.status == :error))
        post_doctor = Enum.find(results, &(&1.step == :post_doctor))
        code = Map.get(post_doctor || %{}, :post_doctor_code, 0)

        code =
          cond do
            failures != [] and code == 0 -> 2
            Enum.any?(failures, &(&1.step in [:pre_doctor, :post_doctor, :hierarchy])) -> 1
            true -> code
          end

        %{
          results: results, exit_code: code, failures: failures,
          next_steps: build_next_steps(results)
        }
      end

      # Q-A2: print Ollama UDS hint as "next step" (resolves Open Question #1)
      defp build_next_steps(_results) do
        [
          "To run offline LLM inference after init:",
          "  1. mkdir -p /tmp && OLLAMA_HOST=unix:///tmp/ollama.sock ollama serve &",
          "  2. ollama pull llama3.2:1b",
          "  3. See .planning/phases/02-.../EXAMPLE_COMPANY_README.md for more."
        ]
      end
    end
